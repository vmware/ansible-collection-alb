Examples
--------

.. code-block:: yaml

    - name: Deploy Controller
      hosts: localhost
      connection: local
      collections:
        - vmware.alb
      vars:
        avi_credentials:
          username: "avi_user"
          password: "avi_password"
          controller: "192.168.138.18"
          api_version: "21.1.1"
          tenant: "admin"
      tasks:
        - name: Example to create ErrorPageProfile object
          avi_errorpageprofile:
            avi_credentials: "{{ avi_credentials }}"
            state: present
            name: sample_errorpageprofile
