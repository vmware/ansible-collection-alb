      vars:
        avi_credentials:
          username: "avi_user"
          password: "avi_password"
          controller: "192.168.138.18"
          api_version: "21.1.1"
      tasks:
        - name: Example to create WafProfile object
          avi_wafprofile:
            avi_credentials: "{{ avi_credentials }}"

            name: sample_wafprofile
