Synopsis

Examples
--------

.. code-block:: yaml

    - name: Deploy Controller
      hosts: localhost
      connection: 
      collections:
        - vmware.alb
      vars:
        avi_credentials:
          username: "avi_user"
          password: "avi_password"
          controller: "192.168.138.18"
          api_version: "21.1.1"
      tasks:
        - name: Example to create SnmpTrapProfile object
          avi_snmptrapprofile:
            avi_credentials: "{{ avi_credentials }}"
            state: present
            name: sample_snmptrapprofile
